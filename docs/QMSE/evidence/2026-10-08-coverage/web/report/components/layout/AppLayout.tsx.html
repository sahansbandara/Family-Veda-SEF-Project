  useEffect(() => {
    const updateIndicator = () => {
      if (!navRef.current) return
      
      // Look for the active class first
      let activeTab = navRef.current.querySelector('a.active') as HTMLElement
      
      // Fallback: match href to location
      if (!activeTab) {
        const tabs = Array.from(navRef.current.querySelectorAll('a'))
        activeTab = tabs.find(tab => tab.getAttribute('href') === location.pathname) as HTMLElement
      }
      
      if (activeTab) {
        setIndicatorStyle({
          left: 0,
          transform: `translateX(${activeTab.offsetLeft}px)`,
          width: activeTab.offsetWidth,
          opacity: 1
        })
      }
    }
    // Update once immediately (DOM is updated by React)
    updateIndicator()
    
    // Fallback for fonts/icons loading causing layout shifts
    const raf = requestAnimationFrame(() => {
      requestAnimationFrame(updateIndicator)
    })
    
    return () => cancelAnimationFrame(raf)
  }, [location.pathname])

  const [theme, setTheme] = useState<'light' | 'dark'>(() => {
