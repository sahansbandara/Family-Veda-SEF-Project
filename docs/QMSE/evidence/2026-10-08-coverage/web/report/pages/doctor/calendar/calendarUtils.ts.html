export type StatusTone = 'primary' | 'warning' | 'success' | 'danger' | 'muted'

/** Cancelled and missed visits stay on the calendar but do not count as visits. */
export const isActive = (appointment: AppointmentDto) => appointment.status !== 'Cancelled' && appointment.status !== 'NoShow'
 
const pad = (value: number) => String(value).padStart(2, '0')
 
/** Local calendar date, never the UTC prefix of the ISO string. */
export const dayKey = (date: Date) => `${date.getFullYear()}-${pad(date.getMonth() + 1)}-${pad(date.getDate())}`
 
export const startOfDay = (date: Date) => new Date(date.getFullYear(), date.getMonth(), date.getDate())
 
export const addDays = (date: Date, days: number) => new Date(date.getFullYear(), date.getMonth(), date.getDate() + days)
 
/** Same day in another month, clamped so 31 January + 1 month is 28 or 29 February. */
export const addMonths = (date: Date, months: number) => {
  const lastDay = new Date(date.getFullYear(), date.getMonth() + months + 1, 0).getDate()
  return new Date(date.getFullYear(), date.getMonth() + months, Math.min(date.getDate(), lastDay))
}
 
export const weekDays = (date: Date) => {
  const sunday = addDays(date, -date.getDay())
  return Array.from({ length: 7 }, (_, index) => addDays(sunday, index))
}
 
/** Whole weeks covering the month of the given date, Sunday first. */
export const monthGrid = (date: Date) => {
  const first = new Date(date.getFullYear(), date.getMonth(), 1)
  const last = new Date(date.getFullYear(), date.getMonth() + 1, 0)
  const start = addDays(first, -first.getDay())
  const cells = last.getDate() + first.getDay()
  const total = Math.ceil(cells / 7) * 7
  return Array.from({ length: total }, (_, index) => addDays(start, index))
}

export const endsAt = (appointment: Pick<AppointmentDto, 'startsAt' | 'durationMinutes'>) =>
  new Date(new Date(appointment.startsAt).getTime() + appointment.durationMinutes * 60_000)
 
export const groupByDay = (appointments: AppointmentDto[]) => {
  const map = new Map<string, AppointmentDto[]>()
  for (const appointment of [...appointments].sort((a, b) => a.startsAt.localeCompare(b.startsAt))) {
    const key = dayKey(new Date(appointment.startsAt))
    map.set(key, [...(map.get(key) ?? []), appointment])
  }
  return map
}
 
/** 24-hour clock: compact enough for the timeline gutter on a phone and unambiguous for a schedule. */
export const formatTime = (date: Date) => date.toLocaleTimeString(undefined, { hour: '2-digit', minute: '2-digit', hourCycle: 'h23' })
 
export const formatLongDay = (date: Date) =>
  date.toLocaleDateString(undefined, { weekday: 'long', day: 'numeric', month: 'long', year: 'numeric' })
 
export const formatShortDay = (date: Date) => date.toLocaleDateString(undefined, { day: 'numeric', month: 'short' })
 
export const rangeTitle = (date: Date, view: CalendarView) => {
  if (view === 'month') return date.toLocaleDateString(undefined, { month: 'long', year: 'numeric' })
  if (view === 'day') return formatLongDay(date)
  const days = weekDays(date)
  return `${formatShortDay(days[0])} – ${formatShortDay(days[6])} ${days[6].getFullYear()}`
}
 
export const initials = (name: string) =>
  name.split(/\s+/).filter(Boolean).slice(0, 2).map((part) => part[0]?.toUpperCase() ?? '').join('') || '?'
 
const weekDayNames: WeekDay[] = ['Sunday', 'Monday', 'Tuesday', 'Wednesday', 'Thursday', 'Friday', 'Saturday']
 
/** Weekly hours for one date, as the clinic-local "HH:mm – HH:mm" strings the backend stores. */
export const workingHoursFor = (date: Date, windows: AvailabilityWindowDto[]) =>
  windows
    .filter((window) => window.dayOfWeek === weekDayNames[date.getDay()])
    .sort((a, b) => a.startTime.localeCompare(b.startTime))
    .map((window) => `${window.startTime.slice(0, 5)} – ${window.endTime.slice(0, 5)}`)
