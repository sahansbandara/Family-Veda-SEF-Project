function isMinor(member: MemberDto): boolean {
  const dob = new Date(member.dateOfBirth)
  if (Number.isNaN(dob.getTime())) return false
  const cutoff = new Date()
  cutoff.setUTCFullYear(cutoff.getUTCFullYear() - 18)
  return dob > cutoff
}

/** Mutually exclusive buckets: every appointment lands in exactly one tab. */
function appointmentTab(appointment: AppointmentDto, now: number = Date.now()): ApptTab {
  const future = new Date(appointment.startsAt).getTime() >= now
  if (future && appointment.status === 'Confirmed') return 'upcoming'
  if (future && appointment.status === 'Requested') return 'requests'
  return 'past'
}
 
function initials(name: string): string {
  const parts = name.replace(/^dr\.?\s+/i, '').split(/\s+/).filter(Boolean)
  return (parts.slice(0, 2).map((part) => part[0]).join('') || '?').toUpperCase()
}

                          <FriendlyStatusBadge status={appointment.status} />
                          <div className="appt-item__actions">
                            <button
                              type="button"
                              className="button button--secondary"
                              aria-expanded={open}
                              aria-controls={`appt-detail-${appointment.id}`}
                              onClick={() => setExpandedId(open ? null : appointment.id)}
                            >
                              {open ? 'Hide' : 'View'}
                            </button>
                            {(appointment.status === 'Requested' || appointment.status === 'Confirmed') && (
                              <button className="button button--danger" type="button" onClick={() => void cancel(appointment)}>
                                Cancel
                              </button>
                            )}
                          </div>
                        </div>
                      </div>
                      {open && (
                        <dl className="appt-detail" id={`appt-detail-${appointment.id}`}>
                          <div><dt>Reason</dt><dd>{appointment.reason || '—'}</dd></div>
                          <div><dt>Duration</dt><dd>{appointment.durationMinutes} minutes</dd></div>
                          <div><dt>Doctor's note</dt><dd>{appointment.doctorNote || 'No note yet'}</dd></div>
                          <div><dt>Requested</dt><dd>{formatDateTime(appointment.createdAt)}</dd></div>
                        </dl>
                      )}
                    </li>
                  )
                })}
              </ul>
            )}
          </div>
        </section>
      </div>
    </div>
  )
}
